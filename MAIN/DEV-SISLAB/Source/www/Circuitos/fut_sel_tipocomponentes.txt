<script language="JavaScript">
function DadosTipo(eu)
{	var f = document.all;
	if( eu.value == "" )
	{
		alert( "Nenhum tipo de componente selecionado" );
		f.caracteristica.style.display = "none";
		f.botaoconfirma.style.display = "none";
	}
	else
	{
		f.caracteristica.style.display = "block";
		f.caracteristica.src = "fut_cad_caracteristicas.asp?tipo=" + eu.value;
		f.botaoconfirma.style.display = "block";
	}
}
function CadastraTipoCarac()	// liga as caracteristicas ao tipo
{
	var car = "", qtd = "", i, fc = window.frames["caracteristica"].document.formulario;

	for( i=0; i<fc.carac_comp.length; i++)
	{
		car += fc.carac_comp[i].value + ",";
		qtd += ExtraiQtdeSemPar(fc.carac_comp[i].text) + ",";
	}

	document.formulario.lista_car.value = car.substring(0, car.length-1);
	document.formulario.lista_qtd.value = qtd.substring(0, qtd.length-1);
	document.formulario.action = "ins_tipocaracteristicas.asp";
	document.formulario.target = "escondido";
	document.formulario.submit(); 
}
function ApagaTipo()
{	var eu = document.formulario.tipocomponente;
	if( eu.value == "" )
		alert( "Nenhum tipo de componente selecionado" );
	else
		alert( "Terminar o apagar");
}
